Metal Slug Vertical Slice

A playable slice of Metal Slug's first mission, written in plain C++ for my Year 1, Block A assignment at BUas. There is no engine underneath: the tmpl8 template gives me a window, a pixel buffer and a basic sprite blit, and the animation, collision, camera, enemies and game flow are built on top of that. It runs from title screen to level-complete, with a HUD, lives, score, a 12-round magazine with a reload, two enemy types and prisoners to rescue.

01

Two-track animation

Marco is drawn as two sprites, legs and torso. The legs track follows movement (idle, walk, jump, crouch, crawl) and the body track follows actions (shoot, reload), so he can fire while walking or in mid-air without a separate sheet for every combination. When no action is playing, the body picks up the legs' frame and timer so the two halves stay in step.

Joining the halves is measured rather than guessed. At load, Animation::Load scans each legs frame for its first row of pixels, the waistline, and stores how far to lift the torso above it. The somersault jump is the exception: the hip stops being a horizontal line, so those six frames use hand-tuned offsets instead.

void Player::UpdateAnimation( float dt, BulletPool& bullets )
{
	const bool moving = Moving();

	PlayerState wantLegs;
	if (!m_grounded)  wantLegs = JUMP;                        // cannot crouch in mid-air
	else if (m_down)  wantLegs = moving ? CRAWL : CROUCH;
	else              wantLegs = moving ? WALK : IDLE;

	if (wantLegs != m_legsState) { m_legsState = wantLegs; m_legsFrame = 0; m_legsTimer = 0; }

	if (!m_bodyBusy)
	{
		PlayerState wantBody = m_legsState;
		if (wantBody != m_bodyState)
		{
			m_bodyState = wantBody;
			m_bodyFrame = m_legsFrame;
			m_bodyTimer = m_legsTimer;
		}
	}

	// bodyBusy is what limits the fire rate to one shot per animation.
	if (!m_bodyBusy && m_fire && m_ammo >0)
	{
		m_bodyState = Crouching() ? CROUCH_SHOOT : SHOOT;
		m_bodyFrame = 0; m_bodyTimer = 0; m_bodyBusy = true;
		Fire( bullets );
	}

	if (!m_bodyBusy && m_ammo == 0)
	{
		m_bodyState = RELOAD;
		m_bodyFrame = 0; m_bodyTimer = 0; m_bodyBusy = true;
	}
	const Animation& la = m_anims[m_legsState];
	m_legsTimer += dt;
	while (m_legsTimer >= 1.0f / la.Fps())
	{
		// Subtract rather than zero, so leftover time carries over.
		m_legsTimer -= 1.0f / la.Fps();
		if (la.Loops()) m_legsFrame = (m_legsFrame + 1) % la.Frames();
		else if (m_legsFrame < la.Frames() - 1) m_legsFrame++;
	}

	const Animation& ba = m_anims[m_bodyState];
	m_bodyTimer += dt;
	while (m_bodyTimer >= 1.0f / ba.Fps())
	{
		m_bodyTimer -= 1.0f / ba.Fps();
		if (ba.Loops()) m_bodyFrame = (m_bodyFrame + 1) % ba.Frames();
		else if (m_bodyFrame < ba.Frames() - 1) m_bodyFrame++;
		// A non-looping animation with nowhere left to go is a finished action.
		else
		{
			if (m_bodyState == RELOAD) m_ammo = MAG_SIZE;
			m_bodyBusy = false;
		}
	}
}

void Player::Draw( Surface* screen, const Camera& cam ) const
{
	const Animation& la = m_anims[m_legsState];
	const Animation& ba = m_anims[m_bodyState];

	const int x = cam.ToScreenX( m_x ), y = cam.ToScreenY( m_y );

	const int flipX = m_facingRight ? 0 : -16 * SCALE;

	const int lf = la.LegsFrames() ? m_legsFrame % la.LegsFrames() : 0;

	if (la.HasLegs())
	{
		Sprite* legs = la.Legs( m_facingRight );
		legs->SetFrame( lf );
		legs->Draw( screen, x + flipX, y );
	}

	Sprite* body = ba.Body( m_facingRight );
	body->SetFrame( m_bodyFrame );

	const bool paired  = (m_bodyState == m_legsState);
	const int  useLift = paired ? la.PairLift( lf ) : la.Lift( lf );
	const int  useOffX = paired ? la.OffX( lf ) : 0;
	const int  bodyDX  = (m_facingRight ? useOffX : -useOffX) * SCALE;

	body->Draw( screen, x + flipX + bodyDX, y - (useLift + ba.BodyLift()) * SCALE );
}
02

Tile collision painted in a PNG

The level's collision is a 260 × 15 image where one pixel is one 16-unit tile: white is solid, green is a one-way platform, black is empty. Changing the level geometry means repainting a pixel, not recompiling.

Movement resolves one axis at a time: move on X and fix X, then apply gravity, move on Y and fix Y. After a diagonal move an overlap can't tell you whether it was a wall or a floor. One-way platforms only catch Marco if his feet were above the surface before the move, and a one-tile step-up lets him walk over small rises instead of stopping dead.

In-game
Player.cppC++
void Player::UpdateMovement( float dt, const Level& level )
{
	const bool  crouching  = Crouching();
	const float MOVE_SPEED = crouching ? CRAWL_SPEED : WALK_SPEED;
	const bool  moving     = Moving();
	const float boxTop     = crouching ? BOX_TOP_CR : BOX_TOP;

	float vx = 0;
	if (m_left)  vx -= MOVE_SPEED;
	if (m_right) vx += MOVE_SPEED;
	m_x += vx * dt;
	if (m_x < MIN_X) m_x = MIN_X;
	if (m_x > MAX_X) m_x = MAX_X;

	// --- resolve x against the tiles ---
	if (vx != 0)
	{
		// Only the leading edge can hit something.
		const float edge = (vx > 0) ? m_x + 31 : m_x + 16;

		bool hit = false;
		for (float y = m_y + boxTop; y < m_y + 48; y += Level::TILE)
			if (level.TileAt( edge, y ) == 1) hit = true;
		if (level.TileAt( edge, m_y + 47 ) == 1) hit = true;

		if (hit)
		{
			bool stepped = false;

			if (m_grounded)
			{
				const int   obsRow = (int)((m_y + 47) / Level::TILE);
				const float obsTop = (float)(obsRow * Level::TILE);
				const float newY   = obsTop - 48;

				bool clear = true;
				for (float y = newY + boxTop; y < newY + 48; y += Level::TILE)
					if (level.TileAt( edge, y ) == 1) clear = false;
				if (level.TileAt( edge, newY + 47 ) == 1) clear = false;

				if (clear && m_y - newY <= Level::TILE) { m_y = newY; stepped = true; }
			}

			if (!stepped)
			{
				const int tcol = (int)(edge / Level::TILE);
				if (vx > 0) m_x = (float)(tcol * Level::TILE) - 32;
				else        m_x = (float)((tcol + 1) * Level::TILE) - 16;
			}
		}
	}

	// --- gravity, then resolve y ---
	m_vy += GRAVITY * dt;
	m_y  += m_vy * dt;

	m_grounded = false;
	const float feet     = m_y + 48;
	const float prevFeet = feet - m_vy * dt;   // where the feet were before this move
	const float boxL     = m_x + 16;
	const float boxR     = m_x + 31;

	if (m_vy >= 0)
	{
		const unsigned char a = level.TileAt( boxL, feet );
		const unsigned char b = level.TileAt( boxR, feet );
		const float tileTop = (float)((int)(feet / Level::TILE) * Level::TILE);

		bool land = (a == 1 || b == 1);

		if (!land && (a == 2 || b == 2) && prevFeet <= tileTop + 1.0f) land = true;

		if (land) { m_y = tileTop - 48; m_vy = 0; m_grounded = true; }
	}

	if (moving) m_facingRight = (vx > 0);
}
Level.cppC++
void Level::Load( const char* tilemapFile )
{
	Surface* tm = new Surface( tilemapFile );

	for (int i = 0; i < COLS * ROWS; i++)
	{
		const uint c = tm->pixels[i];
		const int r = (c >> 16) & 255, g = (c >> 8) & 255, b = c & 255;

		if (g > 150 && r < 100 && b < 100)      m_tiles[i] = 2;   // green -> one-way
		else if (r > 200 && g > 200 && b > 200) m_tiles[i] = 1;   // white -> solid
		else                                    m_tiles[i] = 0;   // black -> empty
	}

	delete tm;
}
03

A camera that ignores jumps

Horizontally the camera just centres on Marco. Vertically it tracks the height of the ground he last stood on, not Marco himself. In the air that value doesn't change, so a jump never moves the view.

On landing, the anchor only moves if the ground height changed by more than 24 units: a hop onto a single 16-unit tile is ignored, a real two-tile climb is followed. The view then eases toward the new height at a fixed rate per second, and hard limits keep Marco in frame whatever the anchor asks for.

In-game
Camera.cppC++
void Camera::Follow( float targetX, float targetY, bool grounded, float dt )
{
	m_x = targetX - VIEW_W / 2;

	if (grounded)
	{
		if (!m_haveGround)
		{
			m_groundY = targetY;
			m_y = targetY - REST_OFFSET;
			m_haveGround = true;
		}
		else if (fabsf( targetY - m_groundY ) > SETTLE)
		{
			m_groundY = targetY;
		}
	}

	if (m_haveGround)
	{
		const float want = m_groundY - REST_OFFSET;
		float t = dt * CATCH_UP;
		if (t > 1.0f) t = 1.0f;
		m_y += (want - m_y) * t;
	}

	if (m_y < targetY - DEAD_BOT) m_y = targetY - DEAD_BOT;
	if (m_y > targetY - DEAD_TOP) m_y = targetY - DEAD_TOP;

	if (m_x < m_minX)          m_x = m_minX;
	if (m_x > m_maxX - VIEW_W) m_x = m_maxX - VIEW_W;
	if (m_y < m_minY)          m_y = m_minY;
	if (m_y > m_maxY - VIEW_H) m_y = m_maxY - VIEW_H;
}
04

Two enemies, one base class

Enemy is an abstract base that owns what every enemy shares: position, health, animation state, hitbox and HoldRange(), which walks toward or away from the player to stay inside a preferred firing band and refuses to step into walls or off ledges. The band has width on purpose. A single threshold would have an enemy flipping between running and shooting every frame.

The rifleman keeps his distance and gives ground when Marco gets close. The shield rebel only advances, and overrides TakeDamage(): bullets from the front are absorbed unless he is in the back half of his own shooting animation, where the sprite shows the shield lowered. That gap is the only way to hit him head-on.

In-game
Enemy.cppC++
Enemy::Approach Enemy::HoldRange( float dt, const Level& level, const Player& player )
{
	const float dx = player.X() - m_x;

	m_facingRight = (dx > 0);

	const float dist = fabsf( dx );
	if (dist > m_wakeRange) return ASLEEP;

	// Too far to shoot without doing it from off screen: close the gap.
	if (dist > m_fireMax)
	{
		const float step = (dx > 0 ? m_walkSpeed : -m_walkSpeed) * dt;
		if (TryMove( level, step )) return CLOSING;

		return IN_RANGE;
	}

	// Uncomfortably close, for an enemy that minds.
	if (m_backsOff && dist < m_fireMin)
	{
		const float step = (dx > 0 ? -m_walkSpeed : m_walkSpeed) * dt;
		if (TryMove( level, step ))
		{
			m_facingRight = (step > 0);
			return BACKING;
		}

		// Cornered. Stand and shoot instead of grinding against the wall.
		return IN_RANGE;
	}

	return IN_RANGE;
}
ShieldRebel.cppC++
bool ShieldRebel::ShieldUp() const
{
	if (m_state != SHOOT) return true;
	return m_frame < SHIELD_DOWN_FRAME;
}

void ShieldRebel::TakeDamage( int amount, bool movingRight )
{
	const bool fromFront = (movingRight != m_facingRight);

	if (fromFront && ShieldUp()) return;      // clang, and nothing else

	Enemy::TakeDamage( amount, movingRight );
}
05

Bullet pools and pixel-perfect hits

Bullets are never allocated during play. They live in a fixed pool of 32 slots: firing revives a dead slot and a hit clears a flag. Player and enemy bullets are two separate pools, so a bullet can never hit its own side and no owner flag is needed.

Touching an enemy uses a two-phase test. A cheap box check rejects almost every pair. Only when the boxes overlap does it scan the rectangle where they intersect for a pixel that is solid in both frames, using the same transparency test the draw call uses, so nothing can collide on a pixel that isn't drawn.

Prisoners use the plain box test: walking into a tied one frees him for points, and he salutes and fades out.

In-game
collision.cppC++
static bool SolidAt(const SpriteMask& m, int x, int y)
{
	const int lx = x - m.x;
	const int ly = y - m.y;

	if (lx < 0 || ly < 0 || lx >= m.frameW || ly >= m.frameH) return false;

	const unsigned int c = m.pixels[ly * m.stride + m.frame * m.frameW + lx];
	return (c & 0xffffff) != 0;
}

bool Overlaps(const SpriteMask& a, const SpriteMask& b)
{
	// Nothing to test against -- an enemy mid-state-change may have no sheet.
	if (!a.pixels || !b.pixels) return false;

	// --- broad phase ---
	const AABB ba = BoxOf(a);
	const AABB bb = BoxOf(b);
	if (!Overlaps(ba, bb)) return false;

	// --- narrow phase ---
	const AABB r = Intersection(ba, bb);

	for (int y = (int)r.y1; y <= (int)r.y2; y++)
		for (int x = (int)r.x1; x <= (int)r.x2; x++)
			if (SolidAt(a, x, y) && SolidAt(b, x, y))
				return true;   // one overlapping pixel is a hit; no need to count them

	return false;
}
BulletPool.cppC++
void BulletPool::Fire( float x, float y, bool toRight )
{
	for (int i = 0; i < MAX; i++)
	{
		if (m_bullets[i].alive) continue;

		m_bullets[i].x = x;
		m_bullets[i].y = y;
		m_bullets[i].vx = toRight ? SPEED : -SPEED;
		m_bullets[i].alive = true;
		return;                 // one bullet, not every free slot
	}
}

void BulletPool::Update( float dt, const Level& level )
{
	for (int i = 0; i < MAX; i++)
	{
		if (!m_bullets[i].alive) continue;

		m_bullets[i].x += m_bullets[i].vx * dt;

		if (m_bullets[i].x < 0 || m_bullets[i].x > Level::WORLD_W) m_bullets[i].alive = false;
		if (level.TileAt( m_bullets[i].x, m_bullets[i].y ) == 1)   m_bullets[i].alive = false;
	}
}

What I learned

Working without an engine meant every convenience was a decision I had to make myself. Keeping gameplay in world units and converting to screen pixels in exactly one place made the camera cheap to add. Deriving values instead of storing them (hitboxes from position, score from the kill and prisoner counters, grounded recomputed every frame) removed a whole class of out-of-sync bugs. If I picked it up again, the first fix would be the placement maths duplicated between Player::Draw and Player::Mask, which should be one shared helper.
